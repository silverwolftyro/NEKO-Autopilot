# --- screen-capture backends (Windows) ---------------------------
# Under hardware-accelerated fullscreen, bitblt / dxcam / mss all return
# black frames. WGC goes through the DWM composition pipeline and is the
# only option that does not touch the registry.
# Missing WGC will not crash: _init_channel marks it unavailable and skips.
# It will, however, silently degrade frame quality and rate.
windows-capture>=2.0.1 ; sys_platform == "win32"
mss>=9.0                    # fallback backend
dxcam>=0.0.5 ; sys_platform == "win32"   # fallback backend
pywin32>=306 ; sys_platform == "win32"   # win32api / win32gui / win32process
